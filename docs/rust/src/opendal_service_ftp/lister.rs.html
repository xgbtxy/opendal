26/// Returns `true` for POSIX `LIST` summary lines such as `"total 0"` or `"total 8"`.
27///
28/// `suppaftp`'s `File::from_str` falls back to `ListParser::parse_mlst`, which
29/// treats arbitrary strings without `=` as filenames, turning `"total 8"` into a
30/// phantom 0-byte file entry.
31fn is_posix_total_line(line: &str) -> bool {
32    let trimmed = line.trim();
33    let Some(rest) = trimmed.strip_prefix("total") else {
34        return false;
35    };
36    let Some(first) = rest.chars().next() else {
37        return false;
38    };
39    if !first.is_ascii_whitespace() {
40        return false;
41    }
42    let count = rest.trim_start();
43    !count.is_empty() && count.bytes().all(|b| b.is_ascii_digit())
44}
45
46/// Parse a single line returned by FTP `LIST`, returning `Ok(None)` for blank
47/// lines, POSIX `"total N"` summary headers, and `"."` / `".."` directory entries.
48pub(crate) fn parse_list_line(line: &str) -> Result<Option<File>> {
49    if line.trim().is_empty() || is_posix_total_line(line) {
50        return Ok(None);
51    }
52
53    let file = File::from_str(line)
54        .map_err(|e| Error::new(ErrorKind::Unexpected, "parse file from response").set_source(e))?;
55
56    if matches!(file.name(), "" | "." | "..") {
57        return Ok(None);
58    }
59
60    Ok(Some(file))
61}

63pub struct FtpLister {
64    path: String,
65    file_iter: IntoIter<String>,
66}
67
68impl FtpLister {
69    pub fn new(path: &str, files: Vec<String>) -> Self {
70        Self {
71            path: path.to_string(),
72            file_iter: files.into_iter(),
73        }
74    }
75}
76
77impl oio::List for FtpLister {
78    async fn next(&mut self) -> Result<Option<oio::Entry>> {
79        let de = loop {
80            let Some(file_str) = self.file_iter.next() else {
81                return Ok(None);
82            };
83            if let Some(de) = parse_list_line(file_str.as_str())? {
84                break de;
85            }
86        };
87
88        let path = self.path.to_string() + de.name();
89
90        let mut meta = if de.is_file() {
91            MetadataBuilder::file(de.size() as u64)
92        } else if de.is_directory() {
93            MetadataBuilder::dir()
94        } else {
95            MetadataBuilder::unknown()
96        };
97        meta.last_modified(Timestamp::try_from(de.modified())?);
98
99        let entry = if de.is_file() {
100            oio::Entry::new(&path, meta.build())
101        } else if de.is_directory() {
102            oio::Entry::new(&format!("{}/", path), meta.build())
103        } else {
104            oio::Entry::new(&path, meta.build())
105        };
106
107        Ok(Some(entry))
108    }
109}

111#[cfg(test)]
112mod tests {
113    use super::*;
114    use opendal_core::raw::oio::List as _;

116    #[tokio::test]
117    async fn test_lister_filters_dots_total_header_and_blank_lines() {
118        let lines = vec![
119            "total 16".to_string(),
120            "   ".to_string(),
121            "drwxr-xr-x 2 user group 4096 Nov  5  2018 .".to_string(),
122            "drwxr-xr-x 4 user group 4096 Nov  5  2018 ..".to_string(),
123            "drwxr-xr-x 2 user group 4096 Nov  5  2018 subdir".to_string(),
124            "-rw-r--r-- 1 user group   12 Nov  5  2018 .hidden".to_string(),
125            "-rw-r--r-- 1 user group   42 Nov  5  2018 total 8".to_string(),
126            "10-19-20  03:19PM       <DIR>          .".to_string(),
127            "10-19-20  03:19PM       <DIR>          ..".to_string(),
128            "10-19-20  03:19PM       <DIR>          dosdir".to_string(),
129            "04-08-14  03:09PM                  128 dos.txt".to_string(),
130        ];
131
132        let mut lister = FtpLister::new("dir/", lines);
133        let mut entries = Vec::new();
134        while let Some(entry) = lister.next().await.unwrap() {
135            entries.push((
136                entry.path().to_string(),
137                entry.metadata().mode(),
138                entry.metadata().content_length(),
139            ));
140        }
141
142        assert_eq!(
143            entries,
144            vec![
145                ("dir/subdir/".to_string(), EntryMode::DIR, 0),
146                ("dir/.hidden".to_string(), EntryMode::FILE, 12),
147                ("dir/total 8".to_string(), EntryMode::FILE, 42),
148                ("dir/dosdir/".to_string(), EntryMode::DIR, 0),
149                ("dir/dos.txt".to_string(), EntryMode::FILE, 128),
150            ]
151        );
152    }
153
154    #[test]
155    fn test_is_posix_total_line() {
156        assert!(is_posix_total_line("total 0"));
157        assert!(is_posix_total_line("total 8"));
158        assert!(is_posix_total_line("  total   1234  "));
159        assert!(!is_posix_total_line("total"));
160        assert!(!is_posix_total_line("total abc"));
161        assert!(!is_posix_total_line("total8"));
162        assert!(!is_posix_total_line(
163            "-rw-r--r-- 1 user group 42 Nov  5  2018 total 8"
164        ));
165    }
