27/// Reader returned by this backend.
28pub struct GcsReader {
29    backend: GcsBackend,
30    // StreamReader opens requests lazily, so keep the per-operation context
31    // available for later range requests.
32    ctx: OperationContext,
33    path: String,
34    args: OpRead,
35}
36
37impl GcsReader {
38    pub(super) fn new(
39        backend: GcsBackend,
40        ctx: OperationContext,
41        path: &str,
42        args: OpRead,
43    ) -> Self {
44        Self {
45            backend,
46            ctx,
47            path: path.to_string(),
48            args,
49        }
50    }
51}
52
53impl oio::StreamRead for GcsReader {
54    async fn open(&self, range: BytesRange) -> Result<(RpRead, Box<dyn oio::ReadStreamDyn>)> {
55        let backend = &self.backend;
56        let path = self.path.as_str();
57        let args = self.args.clone();
58        let error_ctx = ErrorContext::new(ServiceOperation("GetObject"))
59            .with_caller_condition(args.is_conditional());
60        let resp = backend
61            .core
62            .gcs_get_object(&self.ctx, path, range, &args)
63            .await?;
64
65        let status = resp.status();
66
67        let (rp, stream) = match status {
68            StatusCode::OK | StatusCode::PARTIAL_CONTENT => {
69                let mut metadata = parse_into_metadata(path, resp.headers())?.into_builder();
70                // GCS sends the JSON API etag unquoted, as stat and list report
71                // it, and matches only that form in preconditions. Some
72                // emulators quote it.
73                if let Some(etag) = parse_etag(resp.headers())? {
74                    let etag = etag
75                        .strip_prefix('"')
76                        .and_then(|etag| etag.strip_suffix('"'))
77                        .unwrap_or(etag);
78                    metadata.etag(etag);
79                }
80                if let Some(generation) = parse_header_to_str(resp.headers(), X_GOOG_GENERATION)? {
81                    metadata.version(generation);
82                }
83                (RpRead::new(metadata.build()), resp.into_body())
84            }
85            _ => {
86                let (part, mut body) = resp.into_parts();
87                let buf = body.to_buffer().await?;
88                return Err(parse_error(error_ctx, Response::from_parts(part, buf)));
89            }
90        };
91
92        Ok((rp, Box::new(stream) as Box<dyn oio::ReadStreamDyn>))
93    }
94}

112    struct EtagTransport(&'static str);
113
114    impl HttpTransport for EtagTransport {
115        async fn fetch(&self, _: http::Request<Buffer>) -> Result<Response<HttpBody>> {
116            let body = Buffer::from("data");
117            Response::builder()
118                .status(StatusCode::PARTIAL_CONTENT)
119                .header(CONTENT_RANGE, "bytes 0-3/4")
120                .header(ETAG, self.0)
121                .header(X_GOOG_GENERATION, "7")
122                .body(HttpBody::new(stream::iter([Ok(body)]), Some(4)))
123                .map_err(new_request_build_error)
124        }
125    }
126
127    async fn read_etag(header: &'static str) -> Option<String> {
128        let backend = GcsBackend {
129            core: Arc::new(GcsCore {
130                info: ServiceInfo::new("gcs", "/", "test-bucket"),
131                capability: Capability::default(),
132                endpoint: "https://storage.googleapis.com".to_string(),
133                bucket: "test-bucket".to_string(),
134                root: "/".to_string(),
135                sign_ctx: Context::new(),
136                credential_provider: Arc::new(
137                    ProvideCredentialChain::new().push(TokenCredentialProvider::new("test-token")),
138                ),
139                request_signer: Arc::new(RequestSigner::new("storage")),
140                signers: HttpTransportCache::new(),
141                predefined_acl: None,
142                default_storage_class: None,
143                skip_signature: true,
144            }),
145        };
146        let ctx = OperationContext::default()
147            .with_http_transport(HttpTransporter::new(EtagTransport(header)));
148        let reader = GcsReader::new(backend, ctx, "object", OpRead::new());
149        let (rp, _) = reader.open(BytesRange::new(0, Some(4))).await.unwrap();
150        rp.into_metadata()
151            .and_then(|meta| meta.etag().map(str::to_string))
152    }
153
154    #[tokio::test]
155    async fn test_read_etag_matches_json_api_form() {
156        // GCS sends the JSON API resource etag unquoted, as stat and list report it.
157        assert_eq!(
158            read_etag("CKWasoTgyPkCEAE=").await.as_deref(),
159            Some("CKWasoTgyPkCEAE=")
160        );
161        // Emulators such as fake-gcs-server quote it.
162        assert_eq!(
163            read_etag("\"CKWasoTgyPkCEAE=\"").await.as_deref(),
164            Some("CKWasoTgyPkCEAE=")
165        );
166    }
