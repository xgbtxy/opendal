31pub(super) struct GcsGrpcWriter {
32    core: Arc<GcsGrpcCore>,
33    ctx: OperationContext,
34    path: String,
35    args: OpWrite,

37    upload_id: Option<String>,

43impl GcsGrpcWriter {
44    pub(super) fn new(
45        core: Arc<GcsGrpcCore>,
46        ctx: OperationContext,
47        path: &str,
48        args: OpWrite,
49    ) -> Self {
50        Self {
51            core,
52            ctx,
53            path: path.to_string(),
54            args,
55            buffer: None,
56            upload_id: None,
57            write_offset: 0,
58            pending: None,
59            state: GcsGrpcWriterState::Open,
60        }
61    }
62
63    fn error_context(&self, service_operation: ServiceOperation) -> ErrorContext {
64        ErrorContext::new(service_operation).with_if_not_exists(self.args.if_not_exists())
65    }
66
67    fn ensure_open(&self) -> Result<()> {
68        match self.state {
69            GcsGrpcWriterState::Open => Ok(()),
70            GcsGrpcWriterState::Closed => Err(Error::new(
71                ErrorKind::Unexpected,
72                "writer is already closed",
73            )),
74            GcsGrpcWriterState::Aborted => Err(Error::new(
75                ErrorKind::Unexpected,
76                "writer is already aborted",
77            )),
78        }
79    }
80
81    fn write_spec(&self) -> WriteObjectSpec {
82        WriteObjectSpec {
83            resource: Some(Object {
84                name: self.core.object_name(&self.path),
85                bucket: self.core.bucket_resource(),
86                content_type: self.args.content_type().unwrap_or_default().to_string(),
87                content_disposition: self
88                    .args
89                    .content_disposition()
90                    .unwrap_or_default()
91                    .to_string(),
92                content_encoding: self.args.content_encoding().unwrap_or_default().to_string(),
93                cache_control: self.args.cache_control().unwrap_or_default().to_string(),
94                metadata: self
95                    .args
96                    .user_metadata()
97                    .map(|metadata| {
98                        metadata
99                            .into_iter()
100                            .map(|(key, value)| (key.to_owned(), value.to_owned()))
101                            .collect()
102                    })
103                    .unwrap_or_default(),
104                ..Default::default()
105            }),
106            if_generation_match: self.args.if_not_exists().then_some(0),
107        }
108    }
109
110    async fn start_resumable_write(&mut self) -> Result<()> {
111        if self.upload_id.is_some() {
112            return Ok(());
113        }
114
115        let bucket = self.core.bucket_resource();
116        let request = StartResumableWriteRequest {
117            write_object_spec: Some(self.write_spec()),
118        };
119        let request = self
120            .core
121            .request(&self.ctx, request, &[("bucket", &bucket)])
122            .await?;
123        let response = self
124            .core
125            .client()
126            .start_resumable_write(request)
127            .await
128            .map_err(|status| {
129                parse_status(
130                    self.error_context(ServiceOperation("StartResumableWrite")),
131                    status,
132                )
133            })?
134            .into_inner();
135        if response.upload_id.is_empty() {
136            return Err(Error::new(
137                ErrorKind::Unexpected,
138                "GCS started a resumable write without an upload ID",
139            ));
140        }
141        self.upload_id = Some(response.upload_id);
142        Ok(())
143    }
144
145    async fn write_once(&self, body: Buffer) -> Result<Object> {
146        let first_message = write_object_request::FirstMessage::WriteObjectSpec(self.write_spec());
147        let requests = write_object_requests(body, 0, true, first_message)?;
148        let bucket = self.core.bucket_resource();
149        let request = self
150            .core
151            .request(&self.ctx, requests, &[("bucket", &bucket)])
152            .await?;
153        let response = self
154            .core
155            .client()
156            .write_object(request)
157            .await
158            .map_err(|status| {
159                parse_status(self.error_context(ServiceOperation("WriteObject")), status)
160            })?
161            .into_inner();
162        match response.write_status {
163            Some(write_object_response::WriteStatus::Resource(object)) => Ok(object),
164            _ => Err(Error::new(
165                ErrorKind::Unexpected,
166                "GCS write completed without object metadata",
167            )),
168        }
169    }
170
171    async fn finish_pending(&mut self) -> Result<Option<Object>> {
172        let outcome = self.drive_pending().await?;
173        let pending = self
174            .pending
175            .take()
176            .expect("pending upload must exist while finishing");
177        match outcome {
178            UploadOutcome::Persisted(size) if !pending.finish_write => {
179                self.write_offset = size;
180                self.buffer = Some(pending.tail);
181                Ok(None)
182            }
183            UploadOutcome::Finalized(object) if pending.finish_write => {
184                self.write_offset = pending.end_offset()?;
185                self.buffer = None;
186                self.upload_id = None;
187                self.state = GcsGrpcWriterState::Closed;
188                Ok(Some(*object))
189            }
190            UploadOutcome::Persisted(_) => Err(Error::new(
191                ErrorKind::Unexpected,
192                "GCS write did not finalize the object",
193            )),
194            UploadOutcome::Finalized(_) => Err(Error::new(
195                ErrorKind::Unexpected,
196                "GCS finalized an object before close",
197            )),
198        }
199    }
200
201    async fn drive_pending(&mut self) -> Result<UploadOutcome> {
202        loop {
203            if self
204                .pending
205                .as_ref()
206                .expect("pending upload must exist")
207                .in_flight
208            {
209                match self.query_write_status().await? {
210                    QueryOutcome::NotFound => {
211                        return Err(Error::new(
212                            ErrorKind::Unexpected,
213                            "GCS resumable upload no longer exists",
214                        ));
215                    }
216                    QueryOutcome::Persisted(size) => {
217                        if let Some(outcome) = self.reconcile_persisted_size(size)? {
218                            return Ok(outcome);
219                        }
220                    }
221                    QueryOutcome::Finalized(object) => {
222                        if self
223                            .pending
224                            .as_ref()
225                            .expect("pending upload must exist")
226                            .finish_write
227                        {
228                            return Ok(UploadOutcome::Finalized(object));
229                        }
230                        return Err(Error::new(
231                            ErrorKind::Unexpected,
232                            "GCS finalized an object before close",
233                        ));
234                    }
235                }
236            }
237
238            let pending = self.pending.as_ref().expect("pending upload must exist");
239            let body = pending.body.clone();
240            let write_offset = pending.write_offset;
241            let finish_write = pending.finish_write;
242            let upload_id = self
243                .upload_id
244                .clone()
245                .expect("resumable upload ID must exist");
246            self.pending
247                .as_mut()
248                .expect("pending upload must exist")
249                .in_flight = true;
250
251            let first_message = write_object_request::FirstMessage::UploadId(upload_id);
252            let requests =
253                write_object_requests(body.clone(), write_offset, finish_write, first_message)?;
254            let bucket = self.core.bucket_resource();
255            let request = self
256                .core
257                .request(&self.ctx, requests, &[("bucket", &bucket)])
258                .await?;
259            let response = self
260                .core
261                .client()
262                .write_object(request)
263                .await
264                .map_err(|status| {
265                    parse_status(self.error_context(ServiceOperation("WriteObject")), status)
266                })?
267                .into_inner();
268
269            match response.write_status {
270                Some(write_object_response::WriteStatus::PersistedSize(size)) => {
271                    if let Some(outcome) = self.reconcile_persisted_size(size)? {
272                        return Ok(outcome);
273                    }
274                    if finish_write && body.is_empty() {
275                        return Err(Error::new(
276                            ErrorKind::Unexpected,
277                            "GCS did not finalize an empty final write",
278                        ));
279                    }
280                    if size == write_offset {
281                        return Err(Error::new(
282                            ErrorKind::Unexpected,
283                            "GCS write made no progress",
284                        )
285                        .with_context("write_offset", write_offset.to_string())
286                        .with_context("persisted_size", size.to_string())
287                        .set_temporary());
288                    }
289                }
290                Some(write_object_response::WriteStatus::Resource(object)) => {
291                    return Ok(UploadOutcome::Finalized(Box::new(object)));
292                }
293                None => {
294                    return Err(Error::new(
295                        ErrorKind::Unexpected,
296                        "GCS write returned no persisted size or object metadata",
297                    ));
298                }
299            }
300        }
301    }
302
303    async fn query_write_status(&self) -> Result<QueryOutcome> {
304        let upload_id = self
305            .upload_id
306            .as_deref()
307            .expect("resumable upload ID must exist");
308        let request = QueryWriteStatusRequest {
309            upload_id: upload_id.to_string(),
310        };
311        let bucket = self.core.bucket_resource();
312        let request = self
313            .core
314            .request(&self.ctx, request, &[("bucket", &bucket)])
315            .await?;
316        let response = match self.core.client().query_write_status(request).await {
317            Ok(response) => response.into_inner(),
318            Err(status) if status.code() == tonic::Code::NotFound => {
319                return Ok(QueryOutcome::NotFound);
320            }
321            Err(status) => {
322                return Err(parse_status(
323                    ErrorContext::new(ServiceOperation("QueryWriteStatus")),
324                    status,
325                ));
326            }
327        };
328        match response.write_status {
329            Some(query_write_status_response::WriteStatus::PersistedSize(size)) => {
330                Ok(QueryOutcome::Persisted(size))
331            }
332            Some(query_write_status_response::WriteStatus::Resource(object)) => {
333                Ok(QueryOutcome::Finalized(Box::new(object)))
334            }
335            None => Err(Error::new(
336                ErrorKind::Unexpected,
337                "GCS query returned no persisted size or object metadata",
338            )),
339        }
340    }
341
342    fn reconcile_persisted_size(&mut self, size: i64) -> Result<Option<UploadOutcome>> {
343        let pending = self.pending.as_mut().expect("pending upload must exist");
344        let end_offset = pending.end_offset()?;
345        if size < pending.write_offset || size > end_offset {
346            return Err(Error::new(
347                ErrorKind::Unexpected,
348                "GCS returned a persisted size outside the pending write range",
349            )
350            .with_context("write_offset", pending.write_offset.to_string())
351            .with_context("end_offset", end_offset.to_string())
352            .with_context("persisted_size", size.to_string()));
353        }
354
355        let consumed = usize::try_from(size - pending.write_offset).map_err(|err| {
356            Error::new(ErrorKind::Unexpected, "GCS persisted size exceeds usize").set_source(err)
357        })?;
358        if consumed != 0 {
359            pending.body = pending.body.slice(consumed..);
360            pending.write_offset = size;
361        }
362        pending.in_flight = false;
363
364        if pending.body.is_empty() && !pending.finish_write {
365            Ok(Some(UploadOutcome::Persisted(size)))
366        } else {
367            Ok(None)
368        }
369    }
370}

372impl oio::Write for GcsGrpcWriter {
373    async fn write(&mut self, bs: Buffer) -> Result<()> {
374        self.ensure_open()?;
375
376        if let Some(pending) = self.pending.as_ref() {
377            if pending.finish_write {
378                return Err(Error::new(
379                    ErrorKind::Unexpected,
380                    "writer is already finishing",
381                ));
382            }
383            if !buffers_equal(&pending.incoming, &bs) {
384                return Err(Error::new(
385                    ErrorKind::Unexpected,
386                    "writer has an unacknowledged pending write with different data; retry with the same buffer or abort",
387                ));
388            }
389            self.finish_pending().await?;
390            return Ok(());
391        }
392
393        if self.buffer.is_none() && self.upload_id.is_none() {
394            self.buffer = Some(bs);
395            return Ok(());
396        }
397
398        self.start_resumable_write().await?;
399
400        let combined = merge_buffers(self.buffer.clone().unwrap_or_default(), bs.clone());
401        let upload_size = combined.len() / WRITE_CHUNK_ALIGNMENT * WRITE_CHUNK_ALIGNMENT;
402        if upload_size == 0 {
403            self.buffer = Some(combined);
404            return Ok(());
405        }
406
407        let mut body = combined;
408        let tail = body.split_off(upload_size);
409        self.pending = Some(PendingUpload {
410            incoming: bs,
411            body,
412            tail,
413            write_offset: self.write_offset,
414            finish_write: false,
415            in_flight: false,
416        });
417        self.finish_pending().await?;
418        Ok(())
419    }
420
421    async fn close(&mut self) -> Result<Metadata> {
422        self.ensure_open()?;
423
424        if let Some(pending) = self.pending.as_ref() {
425            if !pending.finish_write {
426                return Err(Error::new(
427                    ErrorKind::Unexpected,
428                    "writer has an unacknowledged pending write; retry write or abort before close",
429                ));
430            }
431            let object = self.finish_pending().await?.ok_or_else(|| {
432                Error::new(
433                    ErrorKind::Unexpected,
434                    "GCS write finalized without object metadata",
435                )
436            })?;
437            return Ok(parse_object(&object));
438        }
439
440        if self.upload_id.is_none() {
441            let body = self.buffer.clone().unwrap_or_default();
442            let object = self.write_once(body).await?;
443            self.buffer = None;
444            self.state = GcsGrpcWriterState::Closed;
445            return Ok(parse_object(&object));
446        }
447
448        self.pending = Some(PendingUpload {
449            incoming: Buffer::new(),
450            body: self.buffer.clone().unwrap_or_default(),
451            tail: Buffer::new(),
452            write_offset: self.write_offset,
453            finish_write: true,
454            in_flight: false,
455        });
456        let object = self.finish_pending().await?.ok_or_else(|| {
457            Error::new(
458                ErrorKind::Unexpected,
459                "GCS write finalized without object metadata",
460            )
461        })?;
462        Ok(parse_object(&object))
463    }
464
465    async fn abort(&mut self) -> Result<()> {
466        self.ensure_open()?;
467        if let Some(upload_id) = self.upload_id.as_deref() {
468            let request = CancelResumableWriteRequest {
469                upload_id: upload_id.to_string(),
470            };
471            let bucket = self.core.bucket_resource();
472            let request = self
473                .core
474                .request(&self.ctx, request, &[("bucket", &bucket)])
475                .await?;
476            match self.core.client().cancel_resumable_write(request).await {
477                Ok(_) => {}
478                Err(status) if status.code() == tonic::Code::NotFound => {}
479                Err(status) => {
480                    return Err(parse_status(
481                        ErrorContext::new(ServiceOperation("CancelResumableWrite")),
482                        status,
483                    ));
484                }
485            }
486        }
487        self.buffer = None;
488        self.upload_id = None;
489        self.pending = None;
490        self.state = GcsGrpcWriterState::Aborted;
491        Ok(())
492    }
493}

495struct PendingUpload {
496    incoming: Buffer,
497    body: Buffer,
498    tail: Buffer,
499    write_offset: i64,
500    finish_write: bool,
501    in_flight: bool,
502}
503
504impl PendingUpload {
505    fn end_offset(&self) -> Result<i64> {
506        self.write_offset
507            .checked_add(i64::try_from(self.body.len()).map_err(|err| {
508                Error::new(ErrorKind::Unexpected, "GCS write body exceeds i64").set_source(err)
509            })?)
510            .ok_or_else(|| Error::new(ErrorKind::Unexpected, "GCS write offset exceeds i64"))
511    }
512}

532fn merge_buffers(left: Buffer, right: Buffer) -> Buffer {
533    left.into_iter().chain(right).collect()
534}
535
536fn buffers_equal(left: &Buffer, right: &Buffer) -> bool {
537    use bytes::Buf as _;
538
539    if left.len() != right.len() {
540        return false;
541    }
542    let mut left = left.clone();
543    let mut right = right.clone();
544    while left.has_remaining() {
545        let left_chunk = left.chunk();
546        let right_chunk = right.chunk();
547        let common = left_chunk.len().min(right_chunk.len());
548        if left_chunk[..common] != right_chunk[..common] {
549            return false;
550        }
551        left.advance(common);
552        right.advance(common);
553    }
554    true
555}

557struct WriteObjectRequestState {
558    body: Buffer,
559    write_offset: i64,
560    finish_write: bool,
561    first_message: Option<write_object_request::FirstMessage>,
562    emitted: bool,
563}
564
565fn write_object_requests(
566    body: Buffer,
567    write_offset: i64,
568    finish_write: bool,
569    first_message: write_object_request::FirstMessage,
570) -> Result<impl futures::Stream<Item = WriteObjectRequest> + Send + 'static> {
571    if !finish_write && !body.len().is_multiple_of(WRITE_CHUNK_ALIGNMENT) {
572        return Err(Error::new(
573            ErrorKind::Unexpected,
574            "non-final GCS writes must be aligned to 256 KiB",
575        ));
576    }
577    write_offset
578        .checked_add(i64::try_from(body.len()).map_err(|err| {
579            Error::new(ErrorKind::Unexpected, "GCS write body exceeds i64").set_source(err)
580        })?)
581        .ok_or_else(|| Error::new(ErrorKind::Unexpected, "GCS write offset exceeds i64"))?;
582
583    Ok(stream::unfold(
584        WriteObjectRequestState {
585            body,
586            write_offset,
587            finish_write,
588            first_message: Some(first_message),
589            emitted: false,
590        },
591        |mut state| async move {
592            if state.emitted && state.body.is_empty() {
593                return None;
594            }
595
596            state.emitted = true;
597            let (content, content_len) = if state.body.is_empty() {
598                (None, 0)
599            } else {
600                let size = state.body.len().min(MAX_WRITE_CHUNK_SIZE);
601                (Some(state.body.split_to(size).to_bytes()), size)
602            };
603            let request = WriteObjectRequest {
604                write_offset: state.write_offset,
605                finish_write: state.finish_write && state.body.is_empty(),
606                first_message: state.first_message.take(),
607                data: content.map(|content| {
608                    write_object_request::Data::ChecksummedData(ChecksummedData { content })
609                }),
610            };
611            if content_len != 0 {
612                state.write_offset = state
613                    .write_offset
614                    .checked_add(
615                        i64::try_from(content_len)
616                            .expect("validated GCS write chunk size must fit in i64"),
617                    )
618                    .expect("validated GCS write offset must fit in i64");
619            }
620            Some((request, state))
621        },
622    ))
623}
